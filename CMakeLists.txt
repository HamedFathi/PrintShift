cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PRINTSHIFT_VERSION LIMIT_COUNT 1)
string(STRIP "${PRINTSHIFT_VERSION}" PRINTSHIFT_VERSION)
if(NOT PRINTSHIFT_VERSION MATCHES "^([0-9]+)\\.([0-9]+)\\.([0-9]+)(-[0-9A-Za-z.-]+)?$")
  message(FATAL_ERROR "VERSION must contain a semantic version, for example 1.0.0")
endif()
set(PRINTSHIFT_VERSION_CORE "${CMAKE_MATCH_1}.${CMAKE_MATCH_2}.${CMAKE_MATCH_3}")
set(PRINTSHIFT_VERSION_RC "${CMAKE_MATCH_1},${CMAKE_MATCH_2},${CMAKE_MATCH_3},0")
project(PrintShift VERSION ${PRINTSHIFT_VERSION_CORE} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(PRINTSHIFT_SANITIZERS "Enable address and undefined-behavior sanitizers for engine tests" OFF)
configure_file(cmake/version.hpp.in generated/version.hpp @ONLY)
if(BUILD_TESTING)
  add_executable(engine_tests tests/engine_tests.cpp)
  if(MSVC)
    target_compile_options(engine_tests PRIVATE /W4 /WX /utf-8 /UNDEBUG)
  else()
    target_compile_options(engine_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
    if(PRINTSHIFT_SANITIZERS)
      target_compile_options(engine_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(engine_tests PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
  add_test(NAME engine COMMAND engine_tests)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME project_policy COMMAND "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/scripts/check_project.py")
    add_test(NAME packaging COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s "${PROJECT_SOURCE_DIR}/tests" -p "test_*.py")
  endif()
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Version 1 supports Windows x64. Select the x64 generator/platform.")
  endif()
  enable_language(RC)
  configure_file(cmake/app.manifest.in generated/app.manifest @ONLY)
  configure_file(cmake/app.rc.in generated/app.rc @ONLY)
  add_executable(PrintShift WIN32 src/main.cpp "${PROJECT_BINARY_DIR}/generated/app.rc")
  target_include_directories(PrintShift PRIVATE "${PROJECT_BINARY_DIR}/generated")
  target_compile_definitions(PrintShift PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
  target_link_libraries(PrintShift PRIVATE comctl32 shell32 ole32 uuid uxtheme advapi32 gdi32 user32 wtsapi32)
  if(MSVC)
    target_compile_options(PrintShift PRIVATE /W4 /WX /utf-8 /permissive-)
    set_property(TARGET PrintShift PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_link_options(PrintShift PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT)
  elseif(MINGW)
    target_compile_options(PrintShift PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(PrintShift PRIVATE -municode -static -static-libgcc -static-libstdc++ -Wl,--dynamicbase,--nxcompat)
  endif()
  if(BUILD_TESTING)
    add_test(NAME windows_native_smoke COMMAND PrintShift --self-test)
    set_tests_properties(windows_native_smoke PROPERTIES TIMEOUT 20)
    add_executable(windows_settings_tests tests/windows_settings_tests.cpp)
    target_compile_definitions(windows_settings_tests PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    target_link_libraries(windows_settings_tests PRIVATE advapi32)
    if(MSVC)
      target_compile_options(windows_settings_tests PRIVATE /W4 /WX /utf-8 /UNDEBUG)
    else()
      target_compile_options(windows_settings_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
      if(MINGW)
        target_link_options(windows_settings_tests PRIVATE -static -static-libgcc -static-libstdc++)
      endif()
    endif()
    add_test(NAME windows_capture_setting COMMAND windows_settings_tests)
    set_tests_properties(windows_capture_setting PROPERTIES TIMEOUT 20)
  endif()
  install(TARGETS PrintShift RUNTIME DESTINATION .)
  install(FILES README.md LICENSE CHANGELOG.md DESTINATION .)
endif()
